// Introduce the journey rhythm
Ndef(\journeyRhythm)[1] = \set -> Pbindef(\journeyRhythmSeq);
Ndef(\journeyRhythm).xset(\amp, 0.08);
Ndef(\journeyRhythm).xset(\wet12, 0.4); // Subtle distortion

// Round melody becomes brighter, faster, higher
Pbindef(\roundMelodySeq,
	\dur, Pn(Pseq([0.75, 0.5, 0.25, 1], 1), inf),
	\freq, Pn(Pseq([300, 375, 450, 375, 330, 405, 480, 405], 1), inf),
	\modIndex, Pn(Pseq([2.0, 2.3, 2.1], 1), inf),
	\amp, 0.15
);
Ndef(\roundMelody).xset(\wet10, 0.5, \wet11, 0.4); // More prominent effects

// Soft pad might swell slightly, earth drone recedes further
Ndef(\softPad).xset(\amp, 0.1);
Ndef(\earthDrone).xset(\amp, 0.04);
Ndef(\ancientResonance).xset(\amp, 0.05); // Resonances become very background
)

// === 2.4. Movement III: Myth & Legacy ===
// After approx 60-70 seconds
(
// Swell in the mythic pad
Ndef(\mythicPad).xset(\amp, 0.12, \freq, 60, \atk, 6, \filterCutoff, 1200);
Ndef(\mythicPad).xset(\wet10, 0.9, \wet12, 0.6); // Maxed reverb, slow chorus

// Round melody transitions to grand melody, becoming more expansive
Ndef(\roundMelody)[1] = \set -> Pbindef(\grandMelodySeq); // Swap Pbindef
Ndef(\roundMelody).xset(\amp, 0.2); // Increase amplitude
Ndef(\roundMelody).xset(\modIndex, 3.0); // Increase richness
Ndef(\roundMelody).xset(\wet10, 0.7, \wet11, 0.5); // Prominent effects

// All previous elements converge, carefully balance amplitude
Ndef(\pulseBass).xset(\amp, 0.08);
Ndef(\sparkleSynth).xset(\amp, 0.25);
Ndef(\journeyRhythm).xset(\amp, 0.1);
Ndef(\softPad).xset(\amp, 0.12);

// Earth drone and ancient resonances remain as foundational but background elements
Ndef(\earthDrone).xset(\amp, 0.05);
Ndef(\ancientResonance).xset(\amp, 0.06);
)

// === 2.5. Outro: Enduring Echoes ===
// After approx 90-100 seconds
(
// Rhythmic elements slowly fade out and then stop
Pbindef(\journeyRhythmSeq).stop;
Ndef(\journeyRhythm).fadeTime = 3; Ndef(\journeyRhythm).clear(3);
Pbindef(\sparkleSynthSeq).stop;
Ndef(\sparkleSynth).fadeTime = 3; Ndef(\sparkleSynth).clear(3);
Pbindef(\pulseBassSeq).stop;
Ndef(\pulseBass).fadeTime = 3; Ndef(\pulseBass).clear(3);

// Melodic lines become simpler, sustained, and eventually fade
Pbindef(\grandMelodySeq).stop;
Ndef(\roundMelody).xset(\amp, 0.05, \modIndex, 1.0); // Simplify melody timbre
Ndef(\roundMelody).fadeTime = 5; Ndef(\roundMelody).clear(5);

// Soft pad also fades
Ndef(\softPad).fadeTime = 7; Ndef(\softPad).clear(7);

// Only earth drone and mythic pad remain, slowly diminishing
Ndef(\earthDrone).xset(\amp, 0.03); // Reduce drone volume
Ndef(\mythicPad).xset(\amp, 0.08); // Keep mythic pad slightly higher

// Gradual amplitude fade out for earthDrone and mythicPad
Ndef(\earthDrone).fadeTime = 15;
Ndef(\earthDrone).xset(\amp, 0); // Fade out amplitude
Ndef(\mythicPad).fadeTime = 15;
Ndef(\mythicPad).xset(\amp, 0); // Fade out amplitude

// Final clear for the lingering Ndefs after a long fade
fork {
	15.wait; // Wait for the amplitude fades to mostly complete
	Ndef(\earthDrone).clear(5); // Clear remaining node with a final fade
	Ndef(\mythicPad).clear(5); // Clear remaining node with a final fade
	Ndef(\ancientResonance).clear(5); // Clear any lingering ancient resonance
};
)
